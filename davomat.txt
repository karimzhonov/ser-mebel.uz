  function pointsOf_(r, done) {
    let p = 0;
    if (r.st === 'on_time') { p += PTS.ON_TIME; if (r.in && r.start !== null && r.start !== undefined && parseHM_(r.in) <= r.start - 10) p += PTS.EARLY_BIRD; }
    else if (r.st === 'late') p += r.lr === 'Sababli' ? PTS.LATE_REASON : PTS.LATE_NO_REASON;
    else if (r.st === 'absent') p += PTS.ABSENT;
    else if (r.st === 'noin') p += PTS.NO_IN;
    if (r.in) {
      if (r.uni === 1) p += PTS.UNIFORM_YES; else if (r.uni === 0) p += PTS.UNIFORM_NO;
      if (r.out) p += (r.early || 0) > 0 ? (r.er === 'Sababli' ? PTS.LEFT_EARLY_REASON : PTS.LEFT_EARLY) : PTS.LEFT_OK;
      else if (done) p += PTS.NO_OUT;
    }
    return p;
  }
  function parseDays_(s) { const out = []; String(s || '').replace(/\s/g, '').split(',').forEach(function (p) { const n = parseInt(p, 10); if (n >= 1 && n <= 7 && out.indexOf(n) < 0) out.push(n); }); return out.sort(); }

  function parseDate_(s, today) {
    s = String(s || '').trim().toLowerCase();
    if (s === 'bugun') return today;
    if (s === 'ertaga') return addDays_(today, 1);
    let m = s.match(/^(\d{4})-(\d{2})-(\d{2})$/);
    if (m) return s;
    m = s.match(/^(\d{1,2})[./](\d{1,2})(?:[./](\d{2,4}))?$/);
    if (m) {
      let y = m[3] ? +m[3] : +today.slice(0, 4);
      if (y < 100) y += 2000;
      return y + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[1]).slice(-2);
    }
    return null;
  }

  function findEmp_(q) {
    q = String(q || '').trim().replace(/^#/, '');
    const all = allEmps_();
    if (/^\d+$/.test(q)) return all.filter(function (e) { return e.id === +q; })[0] || null;
    q = q.replace(/^@/, '').toLowerCase();
    const byU = all.filter(function (e) { return e.u && e.u.toLowerCase() === q; });
    if (byU.length) return byU[0];
    const byN = all.filter(function (e) { return e.n.toLowerCase().indexOf(q) >= 0; });
    return byN.length === 1 ? byN[0] : null;
  }

  /** Bugungi (yoki ScriptProperties'da saqlangan) kun uchun har xodim holati. */
  function computeDay_(day, nowD) {
    const since = S.get('since');
    if (since && day < since) return [];
    const deadline = parseHM_(setting_('REPORT_TIME')) || 690;
    const today = dayStr_(nowD);
    const hol = holiday_(day);
    const out = [];
    allEmps_().forEach(function (e) {
      const a = att_(day, e.tg);
      if (!e.a && !(a && (a.in || a.out))) return;
      if (e.c && day < e.c && !a) return;
      let st, note = '';
      if (a && a.in) st = a.st;
      else if (a && a.out) st = 'noin';
      else if (hol) { st = 'holiday'; note = hol; }
      else if (leaveOn_(day, e.tg)) { st = 'leave'; note = leaveOn_(day, e.tg); }
      else if (a && a.st === 'off') { st = 'off'; note = 'tasdiqlangan dam'; }
      else if (empOffDays_(e).indexOf(isoWd_(day)) >= 0) st = 'off';
      else if (a && a.st === 'excused') { st = 'excused'; note = a.rsn || ''; }
      else if (day < today || mod_(nowD) >= deadline) st = 'absent';
      else st = 'waiting';
      out.push({ e: e, st: st, a: a || {}, note: note });
    });
    return out;
  }

  function expectedToday_(e, day) {
    if (holiday_(day) || leaveOn_(day, e.tg)) return false;
    const a = att_(day, e.tg);
    if (a && a.st === 'off') return false;
    return empOffDays_(e).indexOf(isoWd_(day)) < 0;
  }

  function summarize_(rows) {
    const s = { expected: 0, on_time: [], late: [], waiting: [], absent: [], excused: [], rest: [], noUni: [], notConn: [] };
    rows.forEach(function (r) {
      if (s[r.st] && r.st !== 'rest') s[r.st].push(r); else s.rest.push(r);
      if (['on_time', 'late', 'waiting', 'absent', 'excused'].indexOf(r.st) >= 0) s.expected++;
      if (r.a.uni === 0) s.noUni.push(r);
      if (!r.e.dm && r.e.a) s.notConn.push(r.e);
    });
    s.late.sort(function (x, y) { return (y.a.late || 0) - (x.a.late || 0); });
    return s;
  }

  function dayReport_(day, nowD, title) {
    const rows = computeDay_(day, nowD);
    const s = summarize_(rows);
    const L = ['<b>' + title + '</b>', '🗓 ' + fmtDay_(day)];
    if (!s.expected && !s.on_time.length && !s.late.length) {
      L.push('', 'Bugun ishga kutilgan xodim yo\'q (dam / bayram).');
      return L.join('\n');
    }
    const came = s.on_time.length + s.late.length;

    const resting = s.rest.length;

    if (gids.length > 1) {

      gids.forEach(function (gid) {

      });
    }
    if (s.late.length) {

      s.late.forEach(function (r, i) {

        else why = ' — sabab hali aytilmagan';
        L.push((i + 1) + '. ' + esc_(r.e.n) + ' — ' + r.a.in + ' (+' + fmtMin_(r.a.late) + ')' + why);
      });
    }
    if (s.waiting.length) {
      L.push('', '<b>⏳ Hali kelmaganlar:</b>');
      s.waiting.forEach(function (r, i) {

        else if (!r.e.dm) x = ' — botga ulanmagan';
        L.push((i + 1) + '. ' + esc_(r.e.n) + x);
      });
    }
    if (s.absent.length) {
      L.push('', '<b>❌ Kelmaganlar:</b>');
      s.absent.forEach(function (r, i) { L.push((i + 1) + '. ' + esc_(r.e.n) + (r.a.eta ? ' (aytgan vaqti ' + r.a.eta + ')' : '')); });
    }
    if (s.excused.length) {

    }

    if (lv.length) L.push('', "<b>🏖 Ta'tilda:</b> " + lv.map(function (r) { return esc_(r.e.n) + ' (' + esc_(r.note) + ')'; }).join(', '));

    if (s.on_time.length) {
      L.push('', '<b>✅ Vaqtida kelganlar:</b> ' + s.on_time.slice().sort(function (x, y) { return x.a.in < y.a.in ? -1 : 1; })
        .map(function (r) { return esc_(r.e.n) + ' (' + r.a.in + ')'; }).join(', '));
    }
    const yd = addDays_(day, -1);
    let yrows = [];

  const EMP_CMDS = ['holatim', 'kelolmayman', 'ballarim', 'ballar'];
  const ADMIN_CMDS = ['davomat', 'dhisobot', 'dkecha', 'dhafta', 'doy', 'djadval', 'dxodimlar', 'damkun', 'dam', 'tatil', 'tatillar',
    'tatil_ochir', 'bayram', 'bayramlar', 'bayram_ochir', 'dchiqar', 'dqayta', 'dsozlamalar', 'vaqt', 'kechikish', 'dsozla', 'guruhlar', 'dyordam', 'ishvaqti', 'ball', 'reyting', 'dtozala'];

  /**
   * Har bir Telegram yangilanishi avval shu yerga keladi.
   * true = davomatga tegishli, shu yerda bajarildi; false = moliya botiga o'tkaziladi.
   */
  function handle(u) {
    try {
      S.load();
      return route_(u);
    } catch (e) {
      console.error('davomat: ' + (e && e.stack || e));
      return !!(u.callback_query && /^dv:/.test(u.callback_query.data || '')) ||
        !!(u.message && u.message.chat && u.message.chat.type !== 'private');
    }
  }

  function route_(u) {
    if (u.callback_query) {
      if (!/^dv:/.test(u.callback_query.data || '')) return false;
      onCallback_(u.callback_query);
      return true;
    }
    if (u.my_chat_member) { onMyChatMember_(u.my_chat_member); return true; }
    if (u.chat_member) { onChatMember_(u.chat_member); return true; }
    const m = u.message;
    if (!m || !m.from) return false;
    const text = String(m.text || '').trim();
    if (m.chat.type === 'group' || m.chat.type === 'supergroup') {
      // moliya boti guruh xabarlarini ishlatmaydi - hammasi davomatniki
      if (text.charAt(0) === '/') onGroupCommand_(m);
      else if (m.new_chat_members) onNewMembers_(m);
      else if (m.left_chat_member) onLeftMember_(m);
      else onGroupMessage_(m);
      return true;
    }
    if (m.chat.type !== 'private') return false;
    const uid = m.from.id;
    const e = emp_(uid);
    if (e && !e.dm) { e.dm = 1; saveEmp_(e); } // shaxsiyda yozgan bo'lsa, bot unga yoza oladi
    if (text.charAt(0) === '/') {
      const c = parseCmd_(text);
      if (c.cmd === 'start') {
        if (c.args[0] === 'davomat') return employeeStart_(m, true);
        if (!financeUser_(uid)) return employeeStart_(m, false); // moliya botida yo'q, lekin ish guruhida bo'lsa
        return false;
      }
      if (EMP_CMDS.indexOf(c.cmd) >= 0) { onPrivateCommand_(m); return true; }
      if (ADMIN_CMDS.indexOf(c.cmd) >= 0 && isAdmin_(uid)) { adminCommand_(c, uid); return true; }
      return false;
    }
    if (text === MENU_BTN && isAdmin_(uid)) { adminMenu_(uid); return true; }
    return onPrivateText_(m);
  }

  function parseCmd_(text) {
    const parts = String(text).trim().split(/\s+/);
    return { cmd: parts[0].slice(1).split('@')[0].toLowerCase(), args: parts.slice(1) };
  }

  // ---------------- guruh ----------------
  function onGroupMessage_(m) {
    const user = m.from;
    if (user.is_bot || !groups_()[m.chat.id]) return;
    if (isAdmin_(user.id)) return;
    const r = upsertEmp_(user, m.chat.id);
    const e = r.e;
    const at = fromUnix_(m.date);
    const day = dayStr_(at);
    const atMin = mod_(at);
    if (atMin < (parseHM_(setting_('CHECKIN_FROM')) || 300)) return;
    if (m.forward_origin || m.forward_date || m.forward_from) return; // boshqa joydan uzatilgan otchet hisoblanmaydi

    const mode = String(setting_('CHECKIN_MODE')).toLowerCase();
    const isVideoNote = !!m.video_note;
    const hasMedia = !!(m.photo || m.video || m.video_note || m.location || m.venue);
    const txt = String(m.text || m.caption || '').toLowerCase();
    const kws = String(setting_('KEYWORDS')).toLowerCase().split(',').map(function (s) { return s.trim(); }).filter(Boolean);
    const hasKw = kws.some(function (k) { return txt.indexOf(k) >= 0; });
    let ok;
    if (mode === 'video_note') ok = isVideoNote;
    else if (mode === 'media') ok = hasMedia;
    else if (mode === 'keyword') ok = hasKw;
    else if (mode === 'any') ok = hasMedia || !!txt.trim();
    else ok = hasMedia || hasKw;

    if (S.get('since') && day < S.get('since')) return; // hisob hali boshlanmagan
    if (!expectedToday_(e, day)) return;
    const a = att_(day, e.tg) || {};
    if (a.in) {
      if (ok && (mode !== 'any' || hasMedia)) onCheckout_(m, e, day, a, at, atMin);
      return;
    }
    // kelish otcheti yo'q, lekin ish tugashiga 2 soatdan kam qolgan - bu ketish otcheti
    if (ok && (mode !== 'any' || hasMedia) && (a.out || atMin >= empEnd_(e) - 120)) {
      const first = !a.out;
      setAtt_(day, e.tg, { out: hm_(at), noin: 1, wait: '' });
      try { if (setting_('GROUP_ACK') === 'reaction') tg('setMessageReaction', { chat_id: m.chat.id, message_id: m.message_id, reaction: [{ type: 'emoji', emoji: '🫡' }] }); } catch (err) { /* */ }
      if (first) dm_(e, 'Rahmat, ' + ism_(e) + '! Ketish vaqtingiz qayd etildi: <b>' + hm_(at) + '</b>.\n' +
        "Lekin bugun ertalab kelish otchetingizni ko'rmadim. 🤔 Iltimos, ertaga ishga kelganingizda ham guruhga <b>video-doiracha</b> tashlang. Yaxshi dam oling! 🌙");
      return;
    }

    if (!ok) {
      // otchetga o'xshash, lekin noto'g'ri formatda: shaxsiyda bir marta eslatamiz
      if (mode === 'video_note' && (hasMedia || hasKw) && !a.warned) {
        setAtt_(day, e.tg, { warned: 1 });
        dm_(e, ism_(e) + ", otchetingizni ko'rdim, rahmat! 🙂 Faqat bitta iltimos: otchet <b>video-doiracha</b> (dumaloq video) ko'rinishida bo'lishi kerak. " +
          'Guruhga qaytadan dumaloq video tashlab yuboring (Telegramda mikrofon tugmasini bir marta bossangiz, kamera belgisiga o\'tadi).');
      }
      return;
    }

    const start = empStart_(e);
    const late = atMin - start;
    const st = late <= settingInt_('GRACE_MIN') ? 'on_time' : 'late';
    const link = msgLink_(m.chat, m.message_id);
    setAtt_(day, e.tg, { in: hm_(at), late: Math.max(late, 0), st: st, link: link, wait: st === 'late' ? 'lr' : '', ws: hmStr_(start), we: hmStr_(empEnd_(e)) });

    const ack = setting_('GROUP_ACK');
    try {
      if (ack === 'reaction') tg('setMessageReaction', { chat_id: m.chat.id, message_id: m.message_id, reaction: [{ type: 'emoji', emoji: st === 'on_time' ? '🔥' : '✍' }] });
      else if (ack === 'reply') tg('sendMessage', { chat_id: m.chat.id, reply_to_message_id: m.message_id,
        text: st === 'on_time' ? '✅ Qabul qilindi, ' + hm_(at) + ' — vaqtida' : '⏰ Qabul qilindi, ' + hm_(at) + ' — ' + fmtMin_(late) + ' kechikish' });
    } catch (err) { /* muhim emas */ }

    if (st === 'on_time') {
      let t = 'Xayrli tong, ' + ism_(e) + '! ☀️\nOtchetingiz qabul qilindi, kelgan vaqtingiz: <b>' + hm_(at) + '</b>.\n' + pick_(TXT.praise);
      if (settingBool_('UNIFORM_CHECK')) dm_(e, t + '\n\n👕 Bugun formadamisiz?', uniformKb_(day));
      else dm_(e, t);
    } else {
      dm_(e, ism_(e) + ', otchetingiz qabul qilindi, kelgan vaqtingiz: <b>' + hm_(at) + '</b>.\n⏰ Bugun <b>' + fmtMin_(late) + '</b> kechikib keldingiz.\n\n' +
        "Hammasi joyidami? Kechikishga uzrli sabab bo'ldimi?", kb_([[['✅ Sababli', 'lr:' + day + ':1'], ['❌ Sababsiz', 'lr:' + day + ':0']]]));
      if (settingBool_('MANAGER_LIVE_ALERTS')) {
        notifyAdmins_('⏰ <b>' + esc_(e.n) + '</b>' + (groupName_(e.g) ? ' (' + esc_(groupName_(e.g)) + ')' : '') +
          ' kech keldi: ' + hm_(at) + ' (+' + fmtMin_(late) + '). <a href="' + link + '">Otchet</a>');
      }
    }
  }

  /** Kelgandan keyin tashlangan video-doiracha - ketish otcheti */
  function onCheckout_(m, e, day, a, at, atMin) {
    const inMin = parseHM_(a.in);
    if (atMin - inMin < settingInt_('OUT_MIN_GAP')) return; // kelish otchetining takrori
    const end = a.we ? parseHM_(a.we) : empEnd_(e);
    const early = Math.max(end - atMin, 0);
    const isEarly = early > settingInt_('LEAVE_GRACE_MIN');
    const first = !a.out;
    setAtt_(day, e.tg, { out: hm_(at), early: isEarly ? early : 0, er: isEarly ? (a.er || '') : '', wait: isEarly && !a.er ? 'er' : (a.wait === 'er' || a.wait === 'ec' ? '' : a.wait) });
    if (setting_('GROUP_ACK') === 'reaction') {
      try { tg('setMessageReaction', { chat_id: m.chat.id, message_id: m.message_id, reaction: [{ type: 'emoji', emoji: isEarly ? '✍' : '🫡' }] }); } catch (err) { /* */ }
    }
    if (!first && !isEarly) return;
    if (!isEarly) {
      dm_(e, 'Rahmat, ' + ism_(e) + '! 🙏 Ketish vaqtingiz qayd etildi: <b>' + hm_(at) + '</b>.\n' + pick_(TXT.bye));
      return;
    }
    if (a.er) return;
    dm_(e, ism_(e) + ', ketish vaqtingiz qayd etildi: <b>' + hm_(at) + '</b>.\n🚪 Ish vaqtingiz <b>' + hmStr_(end) + '</b> gacha edi, ya\'ni <b>' +
      fmtMin_(early) + '</b> erta ketyapsiz.\n\nRahbar bilan kelishilganmi yoki uzrli sabab bormi?',
      kb_([[['✅ Sababli / kelishilgan', 'er:' + day + ':1'], ['❌ Sababsiz', 'er:' + day + ':0']]]));
  }

  function msgLink_(chat, mid) {
    if (chat.username) return 'https://t.me/' + chat.username + '/' + mid;

  }

  function onNewMembers_(m) {
    if (!groups_()[m.chat.id]) return;
    m.new_chat_members.forEach(function (u) {
      if (u.is_bot || isAdmin_(u.id)) return;
      const r = upsertEmp_(u, m.chat.id);
      if (r.isNew || !r.e.dm) {
        send_(m.chat.id, 'Xush kelibsiz, ' + esc_(r.e.n) + '! 👋\nHar kuni ishga kelganingizda shu guruhga <b>video-doiracha</b> otchet tashlang.\n' +
          "Eslatmalarni olish uchun botga ulaning 👇", { inline_keyboard: [[{ text: '🤖 Botga ulanish', url: botLink_() }]] });
      }
    });
  }

  function onLeftMember_(m) {
    const u = m.left_chat_member;
    if (!groups_()[m.chat.id] || !u || u.is_bot) return;
    const e = emp_(u.id);
    if (e && e.a && String(e.g) === String(m.chat.id)) {
      e.a = 0; saveEmp_(e); S.set('dirty', '1');
      notifyAdmins_('👋 ' + esc_(e.n) + " guruhdan chiqdi va ro'yxatdan olib tashlandi.");
    }
  }

  function onChatMember_(cm) {
    const u = cm.new_chat_member && cm.new_chat_member.user;

    const st = cm.new_chat_member.status;
    if (st === 'left' || st === 'kicked') onLeftMember_({ chat: cm.chat, left_chat_member: u });
  }

    if ((mcm.chat.type === 'group' || mcm.chat.type === 'supergroup') && (st === 'member' || st === 'administrator') &&
        !groups_()[mcm.chat.id]) {
      notifyAdmins_('🤖 Bot "' + esc_(mcm.chat.title) + '" guruhiga qo\'shildi. Kuzatishni boshlash uchun o\'sha guruhda /ulash yozing.');
    }
  }

  function onGroupCommand_(m) {
    const c = parseCmd_(m.text);
    if (c.cmd === 'myid') return send_(m.chat.id, 'Chat ID: <code>' + m.chat.id + '</code>\nSizning ID: <code>' + m.from.id + '</code>');
    if (!isAdmin_(m.from.id)) return;
    if (c.cmd === 'ulash') {
      const g = groups_(); g[m.chat.id] = m.chat.title || String(m.chat.id); S.setJ('groups', g);
      let warn = '';
      try {
        const me = tg('getMe', {});
        if (!me.can_read_all_group_messages) {
          const cmb = tg('getChatMember', { chat_id: m.chat.id, user_id: me.id });
          if (cmb.status !== 'administrator') warn = "\n\n⚠️ Bot barcha xabarlarni ko'rishi uchun uni guruhda <b>admin</b> qiling.";
        }
      } catch (e) { /* */ }
      send_(m.from.id, '✅ "' + esc_(m.chat.title) + '" guruhi kuzatuvga olindi.' + warn);
      send_(m.chat.id, "Hurmatli xodimlar! Har kuni ishga kelganingizda shu guruhga <b>video-doiracha</b> otchet tashlang. " +

  /** "20", "20 daqiqa", "1 soat", "yarim soat", "10:30" -> kelish vaqti (kun daqiqasi) */
  function parseEta_(text, nowMin) {
    const t = String(text || '').toLowerCase().replace(/[‘’`ʻʼ]/g, "'").trim();
    if (!t) return null;
    let m = t.match(/(?:^|\D)(\d{1,2})[:.](\d{2})(?!\d)/);
    if (m && (t.indexOf('soat') < 0 || t.indexOf(':') >= 0)) {
      let tm = (+m[1]) * 60 + (+m[2]);
      if (+m[1] < 24 && +m[2] < 60) {
        if (tm < nowMin - 5 && +m[1] < 12) tm += 720;
        if (tm >= nowMin - 5) return Math.max(tm, nowMin);
      }
    }
    if (/bir yarim soat|1[.,]5 soat|1 yarim/.test(t)) return nowMin + 90;
    if (t.indexOf('yarim soat') >= 0) return nowMin + 30;
    m = t.match(/(\d+(?:[.,]\d+)?)\s*(soat|час|hour)/);
    if (m) return nowMin + Math.round(parseFloat(m[1].replace(',', '.')) * 60);
    m = t.match(/(\d+)\s*(daq|min|м)/);
    if (m) return nowMin + (+m[1]);
    m = t.match(/^\D*(\d{1,3})\D*$/);
    if (m && +m[1] >= 1 && +m[1] <= 240) return nowMin + (+m[1]);
    const words = { "o'n besh": 15, 'yigirma': 20, "o'ttiz": 30, 'ottiz': 30, 'qirq': 40, 'ellik': 50, "o'n": 10, 'bir': 1, 'ikki': 2, 'uch': 3, 'besh': 5 };
    const keys = Object.keys(words);
    for (let i = 0; i < keys.length; i++) {
      if (new RegExp('(^|\\s)' + keys[i] + '(\\s|$)').test(t)) {
        if (t.indexOf('soat') >= 0) return nowMin + words[keys[i]] * 60;
        if (t.indexOf('daq') >= 0 || t.indexOf('min') >= 0) return nowMin + words[keys[i]];
      }
    }
    return null;
  }

  function setSetting_(uid, k, v) {
    v = String(v).trim();
    let val = v, err = null;
    if (['WORK_START', 'WORK_END', 'CHECKIN_FROM', 'REPORT_TIME', 'MONTH_REPORT_TIME'].indexOf(k) >= 0) { const m = parseHM_(v); if (m === null) err = 'vaqt formati 09:00'; else val = hmStr_(m); }
    else if (/_MIN$/.test(k) || k === 'HOLIDAY_ASK_DAYS' || k === 'OUT_MIN_GAP') { const n = parseInt(v, 10); if (isNaN(n) || n < 0 || n > 1440) err = '0..1440 oralig\'ida son'; else val = String(n); }
    else if (k === 'OFF_DAYS') val = parseDays_(v).join(',');
    else if (k === 'CHECKIN_MODE' && ['video_note', 'media', 'keyword', 'media_or_keyword', 'any'].indexOf(v) < 0) err = 'video_note | media | keyword | media_or_keyword | any';
    else if (k === 'GROUP_ACK' && ['reaction', 'reply', 'none'].indexOf(v) < 0) err = 'reaction | reply | none';
    else if (['UNIFORM_CHECK', 'GROUP_REMIND', 'GROUP_LIST_MISSING', 'MANAGER_LIVE_ALERTS'].indexOf(k) >= 0) val = ['1', 'ha', 'yes', 'on', 'true'].indexOf(v.toLowerCase()) >= 0 ? '1' : '0';
    if (err) return send_(uid, "❌ Noto'g'ri qiymat: " + err);
    S.set('s_' + k, val); S.set('dirty', '1');
    send_(uid, '✅ ' + LABELS[k] + ': <b>' + esc_(val) + '</b>');
  }

  function inWin_(nowMin, at, len) { return nowMin >= at && nowMin < at + len; }

  function runSchedule_(nowD) {
    const day = dayStr_(nowD), nm = mod_(nowD), grace = settingInt_('GRACE_MIN');
    const done = [];
    const once = function (n) { const ok = fireOnce_(day, n); if (ok) done.push(n); return ok; };

    // yangi kun: kechagi kunni jadvalga yakuniy yozib, eski ma'lumotni tozalash
    const cur = S.get('curday');
    if (cur && cur !== day) { finalizeDay_(cur); done.push('finalize:' + cur); }
    if (cur !== day) S.set('curday', day);

    const expected = activeEmps_().filter(function (e) { return expectedToday_(e, day); });

    // 1) ertalabki motivatsiya (har guruhga, o'sha guruhdagi eng erta ish boshlanishidan oldin)
    if (settingBool_('GROUP_REMIND') && expected.length) {
      const anyG = expected.some(function (e) { return e.g; });
      Object.keys(groups_()).forEach(function (gid) {
        const ge = anyG ? expected.filter(function (e) { return String(e.g) === gid; }) : expected;
        if (!ge.length) return;
        const starts = ge.map(empStart_).filter(function (x, i, arr) { return arr.indexOf(x) === i; }).sort(function (x, y) { return x - y; });
        if (!inWin_(nm, starts[0] - settingInt_('REMIND_BEFORE_MIN'), 30) || !once('morning_' + gid)) return;
        let t = '☀️ <b>Xayrli tong, jamoa!</b>\n\n' + pick_(TXT.morning) + '\n\n⏰ ' +
          (starts.length === 1 ? 'Ish soat <b>' + hmStr_(starts[0]) + '</b> da boshlanadi. '
            : 'Ish boshlanishi: ' + starts.map(function (x) { return '<b>' + hmStr_(x) + '</b>'; }).join(' / ') + " (har kimning o'z vaqti). ") +
          'Kelganingizda shu yerga <b>video-doiracha</b> tashlang.';
        if (settingBool_('UNIFORM_CHECK')) t += '\n' + pick_(TXT.uniform);
        send_(gid, t);
      });
    }

    const rowsNow = computeDay_(day, nowD);

    // 2) kelmaganlardan shaxsiyda so'rash (har xodimning o'z ish boshlanishidan keyin)
    const missNow = [];
    rowsNow.forEach(function (r) {
      if (r.st !== 'waiting') return;
      const st = empStart_(r.e);
      if (!inWin_(nm, st + grace + 1, 90) || !once('ask_' + r.e.tg)) return;
      missNow.push(r);
      if (dm_(r.e, 'Assalomu alaykum, ' + ism_(r.e) + '! 👋\nIsh soat ' + hmStr_(st) + " da boshlandi, guruhda hali video-doirachangizni ko'rmadim. Hammasi joyidami?\n" +
        'Ishga qancha vaqtda yetib kelasiz? Tugmani bosing yoki yozing (masalan: <i>20 daqiqa</i> yoki <i>10:30</i>).', etaKb_(day))) {
        r.a = setAtt_(day, r.e.tg, { asked: (r.a.asked || 0) + 1 });
      }
    });
    if (missNow.length && settingBool_('GROUP_LIST_MISSING')) {
      Object.keys(groups_()).forEach(function (gid) {
        const names = missNow.filter(function (r) { return String(r.e.g) === gid; }).map(function (r) { return esc_(r.e.u ? '@' + r.e.u : r.e.n); });
        if (names.length) send_(gid, '⏰ Ish boshlandi. Hali otchet tashlamaganlar: ' + names.join(', '));
      });
    }

    // 4) qayta so'rash
    const rep = settingInt_('REPEAT_ASK_MIN');
    if (rep > grace) {
      rowsNow.forEach(function (r) {
        if (r.st !== 'waiting' || r.a.eta || (r.a.asked || 0) >= MAX_ASKS) return;
        if (!inWin_(nm, empStart_(r.e) + rep, 90) || !once('rep_' + r.e.tg)) return;
        if (dm_(r.e, ism_(r.e) + ', hali ham video-doirachangizni ko\'rmadim. 🤔\nHammasi joyidami? Qancha vaqtda yetib kelasiz?', etaKb_(day))) {
          r.a = setAtt_(day, r.e.tg, { asked: (r.a.asked || 0) + 1 });
        }
      });
    }

    // 4b) ish tugadi, ketish otcheti yo'q - eslatish
    rowsNow.forEach(function (r) {
      if (!r.a.in || r.a.out) return;
      const end = r.a.we ? parseHM_(r.a.we) : empEnd_(r.e);
      if (!inWin_(nm, end + settingInt_('OUT_REMIND_MIN'), 120) || !once('outrem_' + r.e.tg)) return;
      dm_(r.e, ism_(r.e) + ', ish kuningiz yakunlandi, rahmat! 🙏\nKetayotganingizda guruhga <b>video-doiracha</b> tashlashni unutmang, ketish vaqtingiz shunda qayd etiladi. 🙂');
    });

    // 5) aytgan vaqti o'tganlar
    computeDay_(day, nowD).forEach(function (r) {
      if (r.st !== 'waiting' || !r.a.eta || r.a.etaSent) return;
      if (nm < parseHM_(r.a.eta) + ETA_GRACE_MIN) return;
      setAtt_(day, r.e.tg, { etaSent: 1 });
      if ((r.a.asked || 0) >= MAX_ASKS) return;
      if (dm_(r.e, ism_(r.e) + ', siz soat ' + r.a.eta + ' da yetib kelaman degan edingiz, lekin guruhda video-doiracha hali yo\'q. ⏳\n' +
        'Yetib kelgan bo\'lsangiz guruhga tashlang. Yana qancha vaqt kerak?', etaKb_(day))) {
        setAtt_(day, r.e.tg, { asked: (r.a.asked || 0) + 1 });
        done.push('eta:' + r.e.tg);
      }
    });

    // 6) kunlik hisobot (11:30)
    if (inWin_(nm, parseHM_(setting_('REPORT_TIME')) || 690, 180) && once('report')) {
      const rows = computeDay_(day, nowD);
      if (expected.length || rows.some(function (r) { return r.a.in; })) {
        notifyAdmins_(dayReport_(day, nowD, '📊 Kunlik hisobot (' + hmStr_(nm) + ')'));
        syncSheet_(nowD);
      }
    }

    // 7) oy oxirida oylik hisobot
    if (addDays_(day, 1).slice(5, 7) !== day.slice(5, 7) && nm >= (parseHM_(setting_('MONTH_REPORT_TIME')) || 1140) && once('month')) {
      syncSheet_(nowD);
      notifyAdmins_(periodReport_(day.slice(0, 8) + '01', day, nowD, '🗓 Oylik hisobot (' + day.slice(0, 7) + ')') +
        "\n\n📊 To'liq jadval: https://docs.google.com/spreadsheets/d/" + book_().getId() + ' ("' + SHEET_NAME + '" listi)');
      const w = monthWinnerMsg_(day.slice(0, 7), nowD);
      notifyAdmins_(w.text, w.kb);
      if (day.slice(5) === '12-31') { const y = yearReport_(day.slice(0, 4), nowD); notifyAdmins_(y.text, y.kb); }
    }

    // 8) bayramdan oldin rahbardan so'rash
    if (nm >= 600) askHolidays_(day).forEach(function (h) { done.push('holask:' + h); });
    return done;
  }

    sh.getRange('I:N').setNumberFormat('@');

  /** computeDay_ qatori uchun ball (bugun bo'lsa, ish tugaganidan 2 soat o'tmaguncha ketish otcheti kutiladi) */
  function rowPoints_(r, day, nowD) {
    const a = r.a;
    const end = a.we ? parseHM_(a.we) : empEnd_(r.e);
    const done = day < dayStr_(nowD) || mod_(nowD) >= end + 120;
    return pointsOf_({ st: r.st, lr: a.lr, in: a.in, start: a.ws ? parseHM_(a.ws) : empStart_(r.e), uni: a.uni, out: a.out, early: a.early, er: a.er }, done);
  }

  function rowsForDay_(day, nowD) {
    return computeDay_(day, nowD).map(function (r) {
      const a = r.a;
      const hours = a.ws ? a.ws + '–' + a.we : empHours_(r.e);
      const scored = ['on_time', 'late', 'absent', 'noin'].indexOf(r.st) >= 0 || a.in;
      return [day, WEEKDAYS[isoWd_(day) - 1], groupName_(r.e.g), r.e.n, r.e.u ? '@' + r.e.u : '', ST_PLAIN[r.st] || r.st,
        a.in || '', r.st === 'late' ? a.late : '', r.st === 'late' ? (a.lr || '') : '', r.st === 'late' ? (a.lc || '') : (r.note || a.rsn || ''),
        !a.in && a.eta ? a.eta : '', a.uni === 1 ? 'Ha' : (a.uni === 0 ? "Yo'q" : ''), String(r.e.tg),
        a.out || '', a.out && a.early ? a.early : '', a.out && a.early ? (a.er || 'Sababsiz') + (a.ec ? ': ' + a.ec : '') : '',
        hours, scored ? rowPoints_(r, day, nowD) : ''];
    });
  }

  /** Kun blokini jadvalga yozadi (bor bo'lsa almashtiradi). O:W ustunlardagi xulosa hisobga olinmaydi. */
  function writeDay_(day, rows) {
    const sh = sheet_();
    const last = sh.getLastRow();
    let first = -1, count = 0, lastLog = 1;
    if (last >= 2) {
      const col = sh.getRange(2, 1, last - 1, 1).getValues();
      for (let i = 0; i < col.length; i++) {
        const v = col[i][0] instanceof Date ? dayStr_(new Date(col[i][0].getTime() + TZ_OFFSET_MIN * 60000)) : String(col[i][0]);
        if (v) lastLog = i + 2;
        if (v === day) { if (first < 0) first = i + 2; count++; }
      }
    }
    if (first > 0) {
      const isLastBlock = first + count - 1 === lastLog;
      if (isLastBlock) {
        sh.getRange(first, 1, count, HEADER.length).clearContent();
        if (rows.length) sh.getRange(first, 1, rows.length, HEADER.length).setValues(rows);
        return;
      }
      sh.deleteRows(first, count);
      if (rows.length) { sh.insertRowsBefore(first, rows.length); sh.getRange(first, 1, rows.length, HEADER.length).setValues(rows); }
      return;
    }
    if (rows.length) sh.getRange(lastLog + 1, 1, rows.length, HEADER.length).setValues(rows);
  }

  function syncSheet_(nowD) {
    const day = dayStr_(nowD);
    writeDay_(day, rowsForDay_(day, nowD));
    writeSummary_(day.slice(0, 7), nowD);
    S.set('dirty', null);
    S.set('synced', String(nowD.getTime()));
  }

  function syncSheetIfDirty_(nowD) {
    if (!S.get('dirty')) return;
    const last = Number(S.get('synced') || 0);
    if (nowD.getTime() - last < 5 * 60000) return; // ko'pi bilan 5 daqiqada bir marta
    syncSheet_(nowD);
  }

  function finalizeDay_(day) {
    // kunni yakuniy holat bilan yozamiz (kelmaganlar "Kelmadi" bo'ladi)
    const endOfDay = new Date(day + 'T23:59:00Z');
    writeDay_(day, rowsForDay_(day, endOfDay));
    writeSummary_(day.slice(0, 7), endOfDay);
    // 3 kundan eski vaqtinchalik ma'lumotni tozalash
    const keep = addDays_(day, -2);
    S.keys('a_').concat(S.keys('f_')).forEach(function (k) { if (k.slice(2, 12) < keep) S.set(k, null); });
  }

  /** Jadvaldan o'tgan kunlar qatorlarini o'qish: {day: [{tg, st, late, lr, uni}]} */
  function readSheet_(from, to) {
    const sh = sheet_();
    const last = sh.getLastRow();
    const out = {};
    if (last < 2) return out;
    const vals = sh.getRange(2, 1, last - 1, HEADER.length).getValues();
    const rev = {}; Object.keys(ST_PLAIN).forEach(function (k) { rev[ST_PLAIN[k]] = k; });
    const t = function (x) { return x instanceof Date ? hm_(new Date(x.getTime() + TZ_OFFSET_MIN * 60000)) : String(x || ''); };
    vals.forEach(function (v) {
      const d = v[0] instanceof Date ? dayStr_(new Date(v[0].getTime() + TZ_OFFSET_MIN * 60000)) : String(v[0]);
      if (d < from || d > to) return;
      const er = String(v[15] || '');
      const r = { tg: Number(v[12]), n: v[3], st: rev[v[5]] || v[5], late: Number(v[7]) || 0, lr: v[8], in: t(v[6]), lc: v[9],
        uni: v[11] === "Yo'q" ? 0 : (v[11] === 'Ha' ? 1 : null), out: t(v[13]), early: Number(v[14]) || 0,
        er: /^Sababli/.test(er) ? 'Sababli' : (er ? 'Sababsiz' : ''), ec: er.replace(/^Sabab(li|siz):?\s*/, ''),
        start: parseHM_(String(v[16] || '').split(/[–-]/)[0]) };
      if (r.start === null || r.start === undefined) r.start = startMin_();
      r.pts = pointsOf_(r, true);
      (out[d] = out[d] || []).push(r);
    });
    return out;
  }

  function lastDayOfMonth_(ym) { return addDays_(addDays_(ym + '-28', 4).slice(0, 8) + '01', -1); }

  function periodStats_(from, to, nowD, pastCache) {
    const today = dayStr_(nowD);
    const past = pastCache || readSheet_(from, to < today ? to : addDays_(today, -1));
    const stats = {};
    const get = function (tgId, name) {
      if (!stats[tgId]) {
        const e = emp_(tgId) || { tg: tgId, n: name, a: 0 };
        stats[tgId] = { e: e, exp: 0, on_time: 0, late: 0, lateR: 0, lateMin: 0, absent: 0, excused: 0, earlyN: 0, viol: 0, worked: 0, pts: 0, adj: 0, bonus: 0 };
      }
      return stats[tgId];
    };
    activeEmps_().forEach(function (e) { get(e.tg, e.n); });
    const add = function (tgId, name, r, pts) {
      const s = get(tgId, name), st = r.st;
      if (['on_time', 'late', 'absent', 'excused'].indexOf(st) >= 0) s.exp++;
      if (st === 'on_time') s.on_time++;
      else if (st === 'late') { s.late++; s.lateMin += r.late || 0; if (r.lr === 'Sababli') s.lateR++; else s.viol++; }
      else if (st === 'absent') { s.absent++; s.viol++; }
      else if (st === 'excused') s.excused++;
      if (r.in || st === 'noin') s.worked++;
      if (st === 'noin') { s.exp++; s.viol++; }
      if (r.out && r.early) { s.earlyN++; if (r.er !== 'Sababli') s.viol++; }
      s.pts += pts || 0;
    };
    Object.keys(past).forEach(function (d) {
      if (d < from || d > to || d >= today) return;
      past[d].forEach(function (r) { add(r.tg, r.n, r, r.pts); });
    });
    if (today >= from && today <= to) computeDay_(today, nowD).forEach(function (r) {
      const a = r.a;
      const scored = ['on_time', 'late', 'absent', 'noin'].indexOf(r.st) >= 0 || a.in;
      add(r.e.tg, r.e.n, { st: r.st, late: a.late, lr: a.lr, in: a.in, out: a.out, early: a.early, er: a.er }, scored ? rowPoints_(r, today, nowD) : 0);
    });
    // rahbarning qo'shimcha ballari
    for (let ym = from.slice(0, 7); ym <= to.slice(0, 7); ym = addDays_(lastDayOfMonth_(ym), 1).slice(0, 7)) {
      S.getJ('adj_' + ym, []).forEach(function (x) {
        if (x.d < from || x.d > to) return;
        const s = get(x.tg, (emp_(x.tg) || {}).n || '?'); s.adj += x.v; s.pts += x.v;
      });
    }
    // to'liq oy bo'lsa - intizom bonusi
    // bonus faqat oyning oxirgi kuni yoki oy tugagach beriladi
    const fullMonth = from.slice(8) === '01' && to === lastDayOfMonth_(from.slice(0, 7)) && today >= to;
    return Object.keys(stats).map(function (k) {
      const s = stats[k];
      s.disc = s.exp ? Math.round(100 * s.on_time / s.exp) : 100;
      if (fullMonth && s.worked && !s.viol) { s.bonus = PTS.MONTH_CLEAN; s.pts += s.bonus; }
      return s;
    }).filter(function (s) { return s.e.a || s.exp; })
      .sort(function (x, y) { return (y.pts - x.pts) || (y.disc - x.disc) || (x.lateMin - y.lateMin) || x.e.n.localeCompare(y.e.n); });
  }
  function monthStats_(ym, nowD, pastCache) { return periodStats_(ym + '-01', lastDayOfMonth_(ym), nowD, pastCache); }

  function periodReport_(from, to, nowD, title) {
    const today = dayStr_(nowD);
    const st = periodStats_(from, to, nowD);
    const L = ['<b>' + title + '</b>', '🗓 ' + from + ' — ' + (to < today ? to : today), ''];
    if (!st.length) return L.concat(["Hali ma'lumot yo'q."]).join('\n');
    const sum = function (f) { return st.reduce(function (a, s) { return a + s[f]; }, 0); };
    const exp = sum('exp'), on = sum('on_time');
    L.push('✅ Vaqtida: <b>' + on + '</b> · ⏰ Kech: <b>' + sum('late') + '</b> (sababli ' + sum('lateR') + ') · ❌ Kelmadi: <b>' + sum('absent') + '</b> · 📝 Sababli: <b>' + sum('excused') + '</b>',
      '📈 Umumiy intizom: <b>' + (exp ? Math.round(100 * on / exp) : 100) + '%</b>',
      '⌛ Jami kechikish: <b>' + fmtMin_(sum('lateMin')) + '</b>', '',
      '<b>Xodimlar reytingi</b> (ball · vaqtida / kech / kelmadi / erta ketdi · intizom):');
    st.forEach(function (s, i) {
      const medal = ['🥇', '🥈', '🥉'][i] || (i + 1) + '.';
      L.push(medal + ' ' + esc_(s.e.n) + ': <b>' + s.pts + '</b> ball · ' + s.on_time + ' / ' + s.late + ' / ' + s.absent + ' / ' + s.earlyN + ' · ' + s.disc + '%' +
        (s.lateMin ? ', kechikish ' + fmtMin_(s.lateMin) : ''));
    });
    const worst = st.filter(function (s) { return s.viol > 0; })
      .sort(function (x, y) { return ((y.late - y.lateR) + 2 * y.absent) - ((x.late - x.lateR) + 2 * x.absent) || y.lateMin - x.lateMin; });
    if (worst.length) L.push('', "<b>⚠️ E'tibor talab qiladi:</b> " + worst.slice(0, 5).map(function (s) {
      return esc_(s.e.n) + ' (' + (s.late - s.lateR) + ' sababsiz kech, ' + s.absent + ' kelmadi' + (s.earlyN ? ', ' + s.earlyN + ' erta ketdi' : '') + ')';
    }).join(', '));
    return L.join('\n');
  }

  function pastDayReport_(day) {
    const rows = readSheet_(day, day)[day] || [];
    const L = ['<b>📊 Hisobot</b>', '🗓 ' + fmtDay_(day), ''];
    if (!rows.length) return L.concat(["Bu kun uchun ma'lumot yo'q."]).join('\n');
    const by = function (st) { return rows.filter(function (r) { return r.st === st; }); };
    const exp = rows.filter(function (r) { return ['on_time', 'late', 'absent', 'excused'].indexOf(r.st) >= 0; }).length;
    L.push('👥 Kutilgan: <b>' + exp + '</b>', '✅ Vaqtida: <b>' + by('on_time').length + '</b>', '⏰ Kech qoldi: <b>' + by('late').length + '</b>',
      '❌ Kelmadi: <b>' + by('absent').length + '</b>', '📝 Sababli kelmadi: <b>' + by('excused').length + '</b>');
    if (by('late').length) {
      L.push('', '<b>⏰ Kech qolganlar:</b>');
      by('late').sort(function (a, b) { return b.late - a.late; }).forEach(function (r, i) {
        L.push((i + 1) + '. ' + esc_(r.n) + ' — ' + r.in + ' (+' + fmtMin_(r.late) + ')' + (r.lr ? ' — ' + r.lr + (r.lc ? ': ' + esc_(r.lc) : '') : ''));
      });
    }
    if (by('absent').length) L.push('', '<b>❌ Kelmaganlar:</b> ' + by('absent').map(function (r) { return esc_(r.n); }).join(', '));
    return L.join('\n');
  }

  }

  function yearStats_(y, nowD) {
    const today = dayStr_(nowD);
    const last = (y + '-12-31') < today ? y + '-12-31' : addDays_(today, -1);
    const past = readSheet_(y + '-01-01', last);
    const tot = {};
    for (let m = 1; m <= 12; m++) {
      const ym = y + '-' + ('0' + m).slice(-2);
      if (ym + '-01' > today) break;
      monthStats_(ym, nowD, past).forEach(function (s) {
        const x = tot[s.e.tg] = tot[s.e.tg] || { e: s.e, pts: 0, exp: 0, on_time: 0, wins: 0, disc: 0, bonus: 0, adj: 0 };
        x.pts += s.pts; x.exp += s.exp; x.on_time += s.on_time; x.bonus += s.bonus; x.adj += s.adj;
        if (Number(S.get('win_m_' + ym)) === Number(s.e.tg)) x.wins++;
      });
    }
    return Object.keys(tot).map(function (k) { const x = tot[k]; x.disc = x.exp ? Math.round(100 * x.on_time / x.exp) : 100; return x; })
      .filter(function (x) { return x.e.a && x.exp; })
      .sort(function (a, b) { return (b.pts - a.pts) || (b.wins - a.wins) || (b.disc - a.disc); });
  }

  function yearReport_(y, nowD) {
    const st = yearStats_(y, nowD);
    const L = ['<b>👑 Yil xodimi tanlovi — ' + y + '</b>', ''];
    if (!st.length) return { text: L.concat(["Hali ma'lumot yo'q."]).join('\n') };
    st.forEach(function (s, i) {
      const medal = ['🥇', '🥈', '🥉'][i] || (i + 1) + '.';
      L.push(medal + ' ' + esc_(s.e.n) + ' — <b>' + s.pts + '</b> ball, intizom ' + s.disc + '%' + (s.wins ? ', 🏆×' + s.wins + ' oy xodimi' : ''));
    });
    L.push('', "Guruhlarda kimni «Yil xodimi» deb e'lon qilamiz?");
    const kb = kb_(st.slice(0, 3).map(function (s, i) { return [[(['🥇', '🥈', '🥉'][i]) + ' ' + s.e.n + " — e'lon qilish", 'ann:y:' + y + ':' + s.e.tg]]; }));
    return { text: L.join('\n'), kb: kb };
  }

  /** Rahbar tanlagan xodimni guruhlarda e'lon qilish */
  function announceWinner_(adminId, kind, period, tgId, edit) {
    const key = 'win_' + kind + '_' + period;
    if (S.get(key)) return edit("Bu tanlov allaqachon e'lon qilingan.");
    const e = emp_(tgId); if (!e) return edit('Xodim topilmadi.');
    const nowD = now_();
    const st = kind === 'm' ? monthStats_(period, nowD) : yearStats_(period, nowD);
    const me = st.filter(function (s) { return s.e.tg === e.tg; })[0] || { pts: 0, disc: 0 };
    S.set(key, String(e.tg));
    const what = kind === 'm' ? '🏆 <b>' + monthName_(period) + ' oyining eng yaxshi xodimi</b>' : '👑 <b>' + period + '-yilning eng yaxshi xodimi</b>';
    const t = '🎉 Hurmatli jamoa!\n\n' + what + ' — <b>' + esc_(e.n) + '</b>! 👏\n' +
      '⭐ ' + me.pts + ' ball · intizom ' + me.disc + '%\n\n' + pick_(TXT.winner) + '\n\nKeyingi g\'olib siz bo\'lishingiz mumkin! 💪';
    const gids = Object.keys(groups_());
    gids.forEach(function (gid) { send_(gid, t); });
    dm_(e, '🎉 ' + ism_(e) + ', tabriklaymiz! Siz ' + (kind === 'm' ? monthName_(period) + ' oyining' : period + '-yilning') +
      ' eng yaxshi xodimi deb topildingiz! 🏆\nIntizomingiz va mehnatingiz uchun katta rahmat. Jamoamiz siz bilan faxrlanadi! 🙏');
    return edit('✅ ' + esc_(e.n) + ' — ' + (kind === 'm' ? 'oy' : 'yil') + " xodimi sifatida " + gids.length + " ta guruhda e'lon qilindi.");
  }

  // =====================================================================
  //  TELEGRAM API
  // =====================================================================
  function tg(method, payload) {
    const res = UrlFetchApp.fetch('https://api.telegram.org/bot' + BOT_TOKEN + '/' + method, {
      method: 'post', contentType: 'application/json', payload: JSON.stringify(payload || {}), muteHttpExceptions: true
    });
    const code = res.getResponseCode();
    let body = {};
    try { body = JSON.parse(res.getContentText()); } catch (e) { /* */ }
    if (code !== 200 || !body.ok) {
      const err = new Error(method + ': ' + code + ' ' + (body.description || ''));
      err.code = code;
      throw err;
    }
    return body.result;
  }

  function send_(chatId, text, kb) {
    const p = { chat_id: chatId, text: text, parse_mode: 'HTML', disable_web_page_preview: true };
    if (kb) p.reply_markup = kb;
    try { return tg('sendMessage', p); } catch (e) { console.warn('send ' + chatId + ': ' + e.message); return null; }
  }

  /** Xodimga shaxsiy xabar. Bot bloklangan bo'lsa false. */
  function dm_(e, text, kb) {
    if (!e.dm) return false;
    const p = { chat_id: e.tg, text: text, parse_mode: 'HTML', disable_web_page_preview: true };
    if (kb) p.reply_markup = kb;
    try { tg('sendMessage', p); return true; } catch (err) {
      if (err.code === 403) { const x = emp_(e.tg); if (x) { x.dm = 0; saveEmp_(x); } e.dm = 0; }
      return false;
    }
  }

  function notifyAdmins_(text, kb) { adminIds_().forEach(function (id) { send_(id, text, kb); }); }

  return { handle: handle, tick: tick, setup: setup, stop: stop, _t: function (src) { return eval(src); } };
})();
