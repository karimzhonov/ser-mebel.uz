function addMonths_(ym, k) {

/** '150000', '150 000', '150k', '150 ming', '1.5 mln', '1,5mln', '2 000 000 so'm', '4200.50' -> son. Xato -> null */
function parseAmount_(text) {
  if (text == null) return null;
  const t = String(text).toLowerCase().replace(/so['ʻ’`]?m|сўм|сум|sum/g, '').trim();
  const m = t.match(/^([\d][\d\s.,]*?)\s*(k|ming|минг|тыс|mln|million|млн|m)?$/);
  if (!m) return null;
  const numS = m[1].trim(), mult = m[2];
  let f = 1, val;
  if (['k', 'ming', 'минг', 'тыс'].indexOf(mult) >= 0) f = 1e3; else if (mult) f = 1e6;
  const ns = numS.replace(/\s/g, '');
  if (f > 1 || /^\d+[.,]\d{1,2}$/.test(ns)) {
    val = parseFloat(ns.replace(',', '.')) * f;
  } else {
    const dg = numS.replace(/[\s.,]/g, '');
    if (!/^\d+$/.test(dg)) return null;
    val = parseInt(dg, 10);
  }
  if (isNaN(val)) return null;
  val = f > 1 || val >= 1000 ? Math.round(val) : Math.round(val * 100) / 100;
  return val > 0 ? val : null;
}
const USD_RE = /\$|usd|dollar|доллар/i;
function parseMoney_(text) {
  const cur = USD_RE.test(text || '') ? 'USD' : 'UZS';
  return { val: parseAmount_(String(text || '').replace(new RegExp(USD_RE.source, 'gi'), ' ').trim()), cur: cur };
}
function findOrder_(text, ids) {
  const known = {};
  ids.forEach(function (id) { known[String(id).toUpperCase()] = id; });
  const toks = String(text || '').split(/[\s,;()]+/);
  for (let tok of toks) {
    tok = tok.toUpperCase().replace(/^#/, '');
    if (!tok) continue;
    if (known[tok]) return known[tok];
    const m = tok.match(/^([A-ZА-ЯЁ]+)-?0*(\d+)$/);
    if (m) {
      const n = m[1] + '-' + String(Number(m[2])).padStart(3, '0');
      if (known[n]) return known[n];
    }
    if (/^\d+$/.test(tok)) {
      for (const k in known) if (/^\d+$/.test(k) && Number(k) === Number(tok)) return known[k];
    }
  }
  return null;
}
/** Tezkor yozuv: '350000 furnitura 406 petlya', '4200$ avans Назира 406' */
function parseQuick_(text, ids) {
  if (!text) return null;
  const t = String(text).trim();
  const lead$ = /^\$/.test(t);
  const m = t.replace(/^\$\s*/, '').match(/^(\d[\d\s.,]*\d|\d)\s*(k|ming|mln|m)?\s*(\$|usd)?\s+(.+)$/i);
  if (!m) return null;
  const val = parseAmount_(m[1] + (m[2] || ''));
  if (!val) return null;
  const rest = m[4].trim();
  const isQoldiq = /qoldiq|остат|доплат|ostatok/i.test(rest);
  const isAvans = !isQoldiq && /avans|аванс|предоплат|oldindan/i.test(rest);
  return {
    kind: (isAvans || isQoldiq) ? 'kirim' : 'chiqim',
    cur: (lead$ || m[3]) ? 'USD' : 'UZS',
    val: val,
    cat: isQoldiq ? 'qoldiq' : isAvans ? 'avans' : guessCat_(rest),
    ord: findOrder_(rest, ids),
    com: rest,
  };
}

function tgFetch_(url, opt) {
  let last;
  for (let i = 0; i < 3; i++) {
    try { return UrlFetchApp.fetch(url, opt); } catch (e) {
      last = e;
      if (!/address unavailable|timeout|timed out|dns|unavailable|bandwidth|service invoked too many/i.test(String(e.message))) break;
      Utilities.sleep(1500 * (i + 1));
    }
  }
  throw new Error(String(last && last.message || last).split(BOT_TOKEN).join('***').replace(/bot\d+:[\w-]+/g, 'bot***'));
}

function startTx_(ctx, kind, target, q) {
  const d = { t: 'tx', kind: kind, uid: String(target), author: ctx.uid, amount: null, cur: 'UZS', orig: null, rate: null,
    cat: null, ord: null, com: null, rcpt: null, oa: false, ca: false, ra: false };

    if (q.cur === 'USD') { d.cur = 'USD'; d.orig = q.val; } else d.amount = Math.round(q.val);
    if (q.cat && !(ADMIN_CATS.indexOf(q.cat) >= 0 && !ctx.admin)) d.cat = q.cat;

}

function orderKb_(kind, cat) {
  // Chiqim — ishdagilar; Avans — yangi (Деталировка) + ishdagilar; Qoldiq — tayyor (Готов, 30 kun)
  let want, general = true;
  if (kind !== 'kirim') want = cat === 'material' ? ['jarayonda', 'yangi'] : ['jarayonda']; // Деталировка zakazlari — faqat «Заказ материал» chiqimida
  else if (cat === 'avans') { want = ['yangi', 'jarayonda', 'kutish']; general = false; }
  else if (cat === 'qoldiq') { want = ['kutish', 'yakun']; general = false; }
  else want = ['yangi', 'jarayonda', 'kutish', 'yakun'];
  const ic = { yangi: '🆕 ', yakun: '✅ ', jarayonda: '🧾 ', kutish: '⏳ ' };
  const list = orders_().filter(function (o) { return want.indexOf(o.status) >= 0; })
    .sort(function (a, b) { return want.indexOf(a.status) - want.indexOf(b.status); });
  const rows = list.slice(0, 60).map(function (o) { return [b_(ic[o.status] + o.id + ' · ' + o.client.slice(0, 28), 'ord:' + o.id)]; });
  if (general || !list.length) rows.push([b_('🏭 Umumiy (buyurtmasiz)', 'ord:-')]);
  if (kind === 'chiqim' && ARX_CATS.indexOf(cat) >= 0) rows.push([b_('🗄 Arxiv ID (Готов, oxirgi 4 oy)', 'oarx:0')]);
  rows.push([CANCEL]);
  return ik_(rows);
}
/** Arxiv: saytda «Готов» bo'lgan buyurtmalar {id: [mijoz, tayyor bo'lgan kun]} — 4 oy saqlanadi */
function arxGet_() { try { return JSON.parse(PropertiesService.getScriptProperties().getProperty('ARXIV') || '{}'); } catch (e) { return {}; } }
function arxList_() {
  const a = arxGet_(), today = today_();
  return Object.keys(a).map(function (id) { return { id: id, client: a[id][0], d: a[id][1] }; })
    .filter(function (x) { return daysBetween_(x.d, today) <= ARX_DAYS; })
    .sort(function (x, y) { return x.d < y.d ? 1 : x.d > y.d ? -1 : (Number(y.id) || 0) - (Number(x.id) || 0); });
}
function arxKb_(page) {
  const L = arxList_(), N = 30, pg = Math.max(0, Math.min(page, Math.ceil(L.length / N) - 1));
  const rows = L.slice(pg * N, pg * N + N).map(function (x) { return [b_('🗄 ' + x.id + ' · ' + String(x.client).slice(0, 24) + ' (' + ddmm_(x.d).slice(0, 5) + ')', 'ord:' + x.id)]; });
  const nav = [];
  if (pg > 0) nav.push(b_('◀️', 'oarx:' + (pg - 1)));
  if ((pg + 1) * N < L.length) nav.push(b_('▶️', 'oarx:' + (pg + 1)));
  if (nav.length) rows.push(nav);
  rows.push([b_('⬅️ Orqaga (faol buyurtmalar)', 'oarx:back')]);
  rows.push([CANCEL]);
  return { kb: ik_(rows), n: L.length, pg: pg, pages: Math.max(1, Math.ceil(L.length / N)) };
}

function isTransfer_(scope, t) {
  if (scope !== 'all') return t.cat === 'otkazma'; // shaxsiy ko'rinishda xodimlar o'rtasidagi o'tkazma ikki tomonda ham hisobga kirmaydi

function expOf_(scope, t) { return t.kind === 'chiqim' && (scope === 'all' ? NON_EXP.indexOf(t.cat) < 0 : t.cat !== 'otkazma'); }
function sideItems_(rows, scope, side) {
  const g = {};
  rows.forEach(function (t) {
    if (side === 'x' ? !expOf_(scope, t) : (t.kind !== 'kirim' || isTransfer_(scope, t))) return;
    g[t.cat] = g[t.cat] || { sum: 0, n: 0 };
    g[t.cat].sum += t.amount; g[t.cat].n++;
  });
  return Object.keys(g).map(function (k) { const c = catInfo_(k); return { code: k, name: c.name, emoji: c.emoji, color: c.color, sum: g[k].sum, n: g[k].n }; })
    .sort(function (a, b) { return b.sum - a.sum; });
}

function recip_(key, hints, knownIds) {
  const ids = String(cfg_()[key] || '').split(/[,\s;]+/).filter(String);
  if (ids.length) return ids;
  const act = users_().filter(function (u) { return u.role === 'xodim' || u.role === 'admin'; });
  const nm = function (u) { return (u.name + ' ' + u.username).toLowerCase().replace(/[^a-z0-9ʻ'а-яё ]+/gi, ' ').replace(/\s+/g, ' '); };
  const byName = act.filter(function (u) { return hints.some(function (h) { return nm(u).indexOf(h) >= 0; }); }).map(function (u) { return u.id; });
  if (byName.length) return byName;
  return act.filter(function (u) { return (knownIds || []).indexOf(u.id) >= 0; }).map(function (u) { return u.id; });
}

function imgSize_(blob) {
  const b = blob.getBytes(), u = function (i) { return b[i] & 255; };
  if (u(0) === 0x89 && u(1) === 0x50) return { w: (u(16) << 24) + (u(17) << 16) + (u(18) << 8) + u(19), h: (u(20) << 24) + (u(21) << 16) + (u(22) << 8) + u(23) };
  if (u(0) === 0xFF && u(1) === 0xD8) {
    let i = 2;
    while (i < b.length - 9) {
      if (u(i) !== 0xFF) { i++; continue; }
      const mk = u(i + 1), len = (u(i + 2) << 8) + u(i + 3);
      if (mk >= 0xC0 && mk <= 0xCF && mk !== 0xC4 && mk !== 0xC8 && mk !== 0xCC) return { h: (u(i + 5) << 8) + u(i + 6), w: (u(i + 7) << 8) + u(i + 8) };
      i += 2 + len;
    }
  }
  return null;
}

function buildHisobTemplate_(blob) {
  const sz = (blob && imgSize_(blob)) || { w: 1225, h: 1178 };
  const W = 918.75, H = Math.round(W * sz.h / sz.w * 100) / 100; // pt (1225 px = 918.75 pt)
  let id = null;
  try { // Slides API: sahifa o'lchami rasmga teng
    const r = UrlFetchApp.fetch('https://slides.googleapis.com/v1/presentations', { method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
      payload: JSON.stringify({ title: 'SerMebel hisob-kitob shabloni', pageSize: { width: { magnitude: W, unit: 'PT' }, height: { magnitude: H, unit: 'PT' } } }) });
    if (r.getResponseCode() === 200) id = JSON.parse(r.getContentText()).presentationId; else console.warn('slides create', r.getContentText().slice(0, 300));
  } catch (e) { console.warn('slides create', e.message); }
  const pres = id ? SlidesApp.openById(id) : SlidesApp.create('SerMebel hisob-kitob shabloni');
  const PW = pres.getPageWidth(), PH = pres.getPageHeight();
  const k = Math.min(PW / sz.w, PH / sz.h), ox = (PW - sz.w * k) / 2, oy = (PH - sz.h * k) / 2, d = k * sz.w / 1225; // dizayn px → pt
  let slide = pres.getSlides()[0];
  if (!slide) slide = pres.appendSlide(SlidesApp.PredefinedLayout.BLANK);
  slide.getPageElements().forEach(function (e) { e.remove(); });
  const AL = { l: SlidesApp.ParagraphAlignment.START, r: SlidesApp.ParagraphAlignment.END, c: SlidesApp.ParagraphAlignment.CENTER };
  const hex = function (c) { return typeof c === 'string' ? c : '#' + c.map(function (x) { return ('0' + x.toString(16)).slice(-2); }).join(''); };
  /** matn qutisi (dizayn px koordinatalarida) */
  const T = function (text, x, y, w, h, size, bold, color, al, italic) {
    const pad = 7.2; // Slides matn qutisining ichki chetini hisobga olamiz
    const sh = slide.insertTextBox(text, ox + x * d - pad, oy + y * d - 3.6, w * d + 2 * pad, h * d + 7.2);
    try { sh.setContentAlignment(SlidesApp.ContentAlignment.MIDDLE); } catch (e) { /* */ }
    try { sh.getAutofit().disableAutofit(); } catch (e) { /* */ }
    const st = sh.getText().getTextStyle().setFontFamily('Montserrat').setFontSize(Math.round(size * d * 10) / 10).setBold(!!bold).setForegroundColor(hex(color));
    if (italic) st.setItalic(true);
    sh.getText().getParagraphStyle().setParagraphAlignment(AL[al || 'l']);
    return sh;
  };
  /** to'rtburchak (fon) */
  const R = function (x, y, w, h, fill, line, round) {
    const sh = slide.insertShape(round ? SlidesApp.ShapeType.ROUND_RECTANGLE : SlidesApp.ShapeType.RECTANGLE, ox + x * d, oy + y * d, w * d, h * d);
    sh.getFill().setSolidFill(fill);
    if (line) { sh.getBorder().getLineFill().setSolidFill(line); sh.getBorder().setWeight(1); } else sh.getBorder().setTransparent();
    return sh;
  };
  if (blob) slide.insertImage(blob, ox, oy, sz.w * k, sz.h * k);
  else drawHisobBase_(T, R);
  Object.keys(HISOB_BOXES).forEach(function (key) {
    const v = HISOB_BOXES[key];
    T(v[8] || '{{' + key + '}}', v[0], v[1], v[2], v[3], v[5], v[6] === 'b', v[7], v[4]);
  });
  pres.saveAndClose();
  setCfg_('HISOB_SHABLON_ID', pres.getId());
  return pres.getId();
}

  }
  if (d.step === 'req_reason') { d.reason = text.slice(0, 300); return reqSend_(ctx, d); }
  if (d.step === 'req_custom') {
    const m = reqMoneyIn_(text);
    if (m.err) return send_(ctx.chat, m.err);
    clrSt_(ctx.uid);
    return reqGive_(ctx, d.req, m);
  }
}
/** So'rov summasi: so'm yoki dollar ('300$', '300 usd') → {amount (so'm), cur, orig, rate} */
function reqMoneyIn_(text) {
  const pm = parseMoney_(text);
  if (!pm.val) return { err: '❗ Summani tushunmadim. Masalan: 500000, 1.5 mln yoki 300$' };
  if (pm.cur !== 'USD') return { amount: Math.round(pm.val), cur: 'UZS', orig: null, rate: null };
  const r = usdRate_();
  if (!r) return { err: "❗ Dollar kursi topilmadi — so'mda yozing yoki rahbar /kurs 12050 qilib kursni kiritsin." };
  return { amount: Math.round(pm.val * r), cur: 'USD', orig: pm.val, rate: r };
}
/** So'rov summasini ko'rsatish: «300 USD (≈ 3 540 000 so'm)» yoki «500 000 so'm» */
function reqMoney_(o) { return o.cur === 'USD' ? usd_(o.orig) + " (≈ " + money_(o.amount) + ')' : money_(o.amount); }

function reqGive_(ctx, id, m) {

function monthForReport_() { const t = today_(); return Number(t.slice(8)) <= 5 ? addMonths_(t.slice(0, 7), -1) : t.slice(0, 7); }

  if (sub) text += '\n\n✅ Bu oy hisobi topshirilgan. Qayta topshirish mumkin.';

function cookiesFrom_(resp, jar) {
  const h = resp.getAllHeaders()['Set-Cookie'];
  [].concat(h || []).forEach(function (c) { const kv = String(c).split(';')[0], i = kv.indexOf('='); if (i > 0) jar[kv.slice(0, i).trim()] = kv.slice(i + 1); });
  return jar;
}
function cookieStr_(jar) { return Object.keys(jar).map(function (k) { return k + '=' + jar[k]; }).join('; '); }
function siteLogin_(sc) {
  if (!sc.login || !sc.pass) throw new Error("Sayt login/paroli kiritilmagan (menyu → «Saytga ulanish»)");
  const url = sc.base + '/admin/login/?next=' + encodeURIComponent(sc.path);
  const r1 = UrlFetchApp.fetch(url, { muteHttpExceptions: true, followRedirects: false });
  const jar = cookiesFrom_(r1, {});
  const m = r1.getContentText().match(/name=["']csrfmiddlewaretoken["']\s+value=["']([^"']+)/) || r1.getContentText().match(/value=["']([^"']+)["']\s+name=["']csrfmiddlewaretoken/);
  if (!m) throw new Error('Login sahifasi topilmadi: ' + url + ' (kod ' + r1.getResponseCode() + ')');
  const r2 = UrlFetchApp.fetch(url, { method: 'post', muteHttpExceptions: true, followRedirects: false,
    headers: { Cookie: cookieStr_(jar), Referer: url },
    payload: { csrfmiddlewaretoken: m[1], username: sc.login, password: sc.pass, next: sc.path } });
  cookiesFrom_(r2, jar);
  if (!jar.sessionid && !Object.keys(jar).some(function (k) { return /session/i.test(k); }))
    throw new Error("Saytga kirib bo'lmadi — login yoki parol noto'g'ri (kod " + r2.getResponseCode() + ')');
  const s = cookieStr_(jar);
  CacheService.getScriptCache().put('site_cookie', s, 18000);
  return s;
}
function isLoginPage_(html) { return /id_password|name=["']password["']/.test(html) && !/result_list/.test(html); }
/** GET sahifa. null — sahifa yo'q (masalan, sahifalash tugadi) */
function siteGet_(sc, url) {
  const cache = CacheService.getScriptCache();
  for (let attempt = 0; attempt < 2; attempt++) {
    const ck = (attempt === 0 && cache.get('site_cookie')) || siteLogin_(sc);
    const r = UrlFetchApp.fetch(url, { muteHttpExceptions: true, followRedirects: false, headers: { Cookie: ck } });
    const code = r.getResponseCode(), html = r.getContentText(), loc = String((r.getAllHeaders() || {}).Location || '');
    if (code === 200 && !isLoginPage_(html)) return html;
    if (code === 302 && loc.indexOf('/login') < 0) return null; // masalan ?e=1 — bunday sahifa yo'q
    if (code === 404) throw new Error("Sahifa topilmadi: " + url + " (Sozlamalar → SAYT_ROYXAT ni tekshiring)");
    if (code === 403) throw new Error("Bu foydalanuvchida «Заказы»ni ko'rish huquqi yo'q");
    cache.remove('site_cookie'); // sessiya tugagan — qayta kiramiz
  }
  throw new Error("Ro'yxat sahifasini ochib bo'lmadi");
}
function htmlText_(s) {
  return String(s).replace(/<br\s*\/?>/gi, ' ').replace(/<[^>]+>/g, '').replace(/&nbsp;/g, ' ').replace(/&amp;/g, '&').replace(/&quot;/g, '"')
    .replace(/&#39;|&#x27;/g, "'").replace(/&lt;/g, '<').replace(/&gt;/g, '>').replace(/&#(\d+);/g, function (_, n) { return String.fromCharCode(n); }).replace(/\s+/g, ' ').trim();
}
/** Django admin (Unfold ham) changelist jadvalini o'qish: [{pk, cells:{maydon: matn}}] */
function parseChangelist_(html, path) {
  const out = { rows: [], cols: [], trCount: 0, sample: '' };
  const i0 = html.search(/id=["']result_list["']/);
  if (i0 < 0) return out;
  let seg = html.slice(i0);
  const pg = seg.search(/class=["'][^"']*paginator/);
  if (pg > 0) seg = seg.slice(0, pg);
  const th = seg.match(/<thead[\s\S]*?<\/thead>/);
  if (th) {
    th[0].replace(/<th[^>]*class=["'][^"']*column-([\w-]+)[^"']*["'][^>]*>([\s\S]*?)<\/th>/g, function (_, f, t) { out.cols.push({ field: f, title: htmlText_(t) }); });
    seg = seg.slice(seg.indexOf(th[0]) + th[0].length);
  }
  seg.replace(/<tr[\s>][\s\S]*?<\/tr>/g, function (tr) {
    out.trCount++;
    if (!out.sample) out.sample = tr.slice(0, 600);
    const cells = {};
    tr.replace(/<(td|th)[^>]*class=["'][^"']*field-([\w-]+)[^"']*["'][^>]*>([\s\S]*?)<\/\1>/g, function (__, tag, f, v) {
      const sel = v.match(/<option[^>]*selected[^>]*>([\s\S]*?)<\/option>/); // tahrirlanadigan ro'yxat bo'lsa — tanlangani
      cells[f] = htmlText_(sel ? sel[1] : v);
    });
    if (!Object.keys(cells).length) return;
    const cb = tr.match(/name=["']_selected_action["'][^>]*value=["']([^"']+)/) || tr.match(/value=["']([^"']+)["'][^>]*name=["']_selected_action/);
    const hm = tr.match(/href=["'][^"']*\/admin\/[\w-]+\/[\w-]+\/([^\/"'?]+)\/(?:change\/?)?["'?]/);
    const pk = (cb && cb[1]) || (hm && hm[1]) || cells.id || cells.order_number || cells.pk;
    if (pk) out.rows.push({ pk: String(pk), cells: cells });
  });
  return out;
}

const RU_MON = ['январ', 'феврал', 'март', 'апрел', 'ма', 'июн', 'июл', 'август', 'сентябр', 'октябр', 'ноябр', 'декабр'];
/** «29 ноября 2026 г.» / «29.11.2026» / «2026-11-29» -> 'yyyy-MM-dd' yoki '' */
function parseRuDate_(s) {
  s = String(s || '').toLowerCase().trim();
  let m = s.match(/(\d{4})-(\d{2})-(\d{2})/);
  if (m) return m[1] + '-' + m[2] + '-' + m[3];
  m = s.match(/(\d{1,2})\.(\d{1,2})\.(\d{4})/);
  if (m) return m[3] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[1]).slice(-2);
  m = s.match(/(\d{1,2})\s+([а-яё]+)\s+(\d{4})/);
  if (m) {
    let mi = -1;
    RU_MON.forEach(function (p, i) { if (m[2].indexOf(p) === 0 && (mi < 0 || p.length > RU_MON[mi].length)) mi = i; });
    if (mi >= 0) return m[3] + '-' + ('0' + (mi + 1)).slice(-2) + '-' + ('0' + m[1]).slice(-2);
  }
  return '';
}

function moneyNum_(t) {
  t = String(t || '').replace(/\u00a0/g, ' ');
  const m = t.match(/-?\d[\d\s.,]*/);
  if (!m) return null;
  let v = m[0].replace(/\s/g, '');
  if (/[.,]\d{1,2}$/.test(v)) { const dec = v.slice(-3).replace(',', '.'); v = v.slice(0, -3).replace(/[.,]/g, '') + (dec.length === 3 ? dec : ''); }
  else v = v.replace(/[.,]/g, '');
  const n = Number(v);
  return isNaN(n) ? null : n;
}

/** Saytdan: to'langan va qoldiq summa. Topilmasa — null (bot o'z yozuvlaridan hisoblaydi) */
function siteMoneyFor_(id, total) {
  if (!PropertiesService.getScriptProperties().getProperty('SAYT_LOGIN')) return null;

  const c = cfg_(), qName = String(c.SAYT_QOLDIQ_MAYDON || '').trim(), pName = String(c.SAYT_TOLANGAN_MAYDON || '').trim();
  const txt = function (x) { return (x.name + ' ' + x.label).toLowerCase(); };
  const pick = function (name, rx, not) {
    const x = f.find(function (y) { return name ? y.name === name : rx.test(txt(y)) && !(not && not.test(txt(y))) && moneyNum_(y.value) !== null; });
    return x ? moneyNum_(x.value) : null;
  };
  let rest = pick(qName, /остат|ostat|qoldiq|долг|remain|balance|debt|qarz/i);
  let paid = pick(pName, /оплачен|оплата|оплат|paid|prepay|предоплат|аванс|avans|внес|to.?lan/i, /итого|total|остат|qoldiq/i);
  if (paid === null) { // inline to'lovlar ro'yxati: summa kataklarini qo'shamiz
    const rows = f.filter(function (y) { return /^(amount|sum|summa|price|payment_amount|value)$/i.test(y.name) && moneyNum_(y.value) !== null; });

  if (paid === null && rest !== null && total) paid = Math.max(0, total - rest);
  if (rest === null && paid === null) return null;
